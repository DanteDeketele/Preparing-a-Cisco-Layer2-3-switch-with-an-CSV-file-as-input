! VLAN Configuration Commands
hostname Switch

ip routing

vlan 10
name Users-on-Switch-1
interface vlan 10
no ip address
no shutdown
exit

interface range Fa0/1 - 12
	switchport access vlan 10
	switchport mode access
	spanning-tree portfast
	no shutdown
	exit

vlan 30
name Servers-on-Switch-1
interface vlan 30
no ip address
no shutdown
exit

interface range Fa0/13 - 16
	switchport access vlan 30
	switchport mode access
	spanning-tree portfast
	no shutdown
	exit

vlan 99
name Management-on-Switch-1
interface vlan 99
ip address 192.168.99.2 255.255.255.0
no shutdown
exit


interface Fa0/23
	description Trunk-on-Switch-1
	switchport mode trunk
	switchport trunk allowed vlan 10-40
	no shutdown
	exit
